package org.howard.edu.lsp.midterm.question4;

// ---------- Interfaces ----------
public interface Networked {
  void connect();
  void disconnect();
  boolean isConnected();
}

public interface BatteryPowered {
  int getBatteryPercent();
  void setBatteryPercent(int percent);
}

// ---------- Abstract Base Class ----------
public abstract class Device {
  private String id;
  private String location;
  private boolean connected;

  // fixed comma and semicolon issues
  public Device(String id, String location) {
    this.id = id;
    this.location = location;
    this.connected = false;
  }

  public String getId() { return id; }
  public String getLocation() { return location; }

  public boolean isConnected() { return connected; }
  protected void setConnected(boolean connected) { this.connected = connected; }

  public void heartbeat() {
    System.out.println("heartbeat");
  }

  // required abstract method
  public abstract String getStatus();
}

// ---------- Concrete Class ----------
public class Camera extends Device implements Networked, BatteryPowered {
  private int batteryPercent;

  // fixed commas and validation
  public Camera(String id, String location, int batteryPercent) {
    super(id, location);
    setBatteryPercent(batteryPercent);
  }

  // Networked
  @Override
  public void connect() { setConnected(true); }

  @Override
  public void disconnect() { setConnected(false); }

  @Override
  public boolean isConnected() { return super.isConnected(); }

  // BatteryPowered
  @Override
  public int getBatteryPercent() { return batteryPercent; }

  @Override
  public void setBatteryPercent(int percent) {
    if (percent < 0 || percent > 100)
      throw new IllegalArgumentException("Battery percent must be 0–100");
    this.batteryPercent = percent;
  }

  // Status
  @Override
  public String getStatus() {
    String conn = isConnected() ? "up" : "down";
    return "Camera[id=" + getId() + ", loc=" + getLocation()
        + ", conn=" + conn + ", batt=" + batteryPercent + "%]";
  }
}
